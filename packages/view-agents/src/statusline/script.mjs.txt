// Claude Code statusLine command installed by Kedalion (one per account, argv[2] = account id).
// Stores the usage JSON Claude Code sends on stdin, then chains the account's previous statusLine command.
// Plain ESM with node builtins only: Kedalion writes this file to `<dataDir>/agents/statusline.mjs` and runs it with bun.
import { spawnSync } from 'node:child_process';
import { mkdirSync, readFileSync, renameSync, writeFileSync } from 'node:fs';
import { homedir } from 'node:os';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';

const CHAIN_TIMEOUT_MS = 5000;

function readStdin() {
  try {
    return readFileSync(0, 'utf8');
  } catch {
    return '';
  }
}

function safeName(s) {
  return String(s).replace(/[^A-Za-z0-9._-]/g, '_');
}

const round1 = (n) => Math.round(n * 10) / 10;
const isNum = (n) => typeof n === 'number' && Number.isFinite(n);

function readJson(file) {
  try {
    const v = JSON.parse(readFileSync(file, 'utf8'));
    return v && typeof v === 'object' ? v : {};
  } catch {
    return {};
  }
}

/** The signed-in account's email from Claude's config (`CLAUDE_CONFIG_DIR/.claude.json`, else `~/.claude.json`). */
function emailOf() {
  const dir = process.env.CLAUDE_CONFIG_DIR;
  const files = dir ? [join(dir, '.claude.json')] : [join(homedir(), '.claude.json')];
  for (const file of files) {
    const e = readJson(file).oauthAccount?.emailAddress;
    if (typeof e === 'string' && e) return e;
  }
  return undefined;
}

/** Effort level as the payload may carry it: a string, or `{ level }`. */
function effortOf(input) {
  const e = input.effort;
  if (typeof e === 'string' && e) return e;
  if (e && typeof e === 'object' && typeof e.level === 'string' && e.level) return e.level;
  return undefined;
}

/**
 * Merges the payload into the previous record. Context % is rounded to 1 decimal; the startup context % is the first reading
 * above 0 and kept forever; model and effort: latest wins; a payload without rate limits never erases the last reading.
 */
function usageOf(input, sessionId, prev) {
  const out = { ...prev, updatedAt: Date.now(), sessionId };
  if (input.model && typeof input.model === 'object') out.model = input.model;
  const effort = effortOf(input);
  if (effort) out.effort = effort;
  if (typeof input.version === 'string' && input.version) out.version = input.version;
  const cw = input.context_window;
  if (cw && typeof cw === 'object') {
    const ctx = { ...cw };
    if (isNum(cw.used_percentage)) ctx.used_percentage = round1(cw.used_percentage);
    out.context_window = ctx;
    if (isNum(ctx.used_percentage) && ctx.used_percentage > 0 && !(prev.startupContextPercent > 0)) out.startupContextPercent = ctx.used_percentage;
  }
  const rl = input.rate_limits;
  if (rl && typeof rl === 'object') {
    const merged = { ...(prev.rate_limits ?? {}) };
    for (const k of ['five_hour', 'seven_day']) if (rl[k] && typeof rl[k] === 'object') merged[k] = rl[k];
    out.rate_limits = merged;
  }
  const email = emailOf();
  if (email) out.email = email;
  return out;
}

function store(dir, account, input) {
  if (!input || typeof input !== 'object' || typeof input.session_id !== 'string' || !input.session_id) return;
  const sessionId = input.session_id;
  const folder = join(dir, 'usage', safeName(account));
  const file = join(folder, `${safeName(sessionId)}.json`);
  mkdirSync(folder, { recursive: true });
  writeFileSync(`${file}.tmp`, JSON.stringify(usageOf(input, sessionId, readJson(file))));
  renameSync(`${file}.tmp`, file);
}

/** `<cwd> · ctx N%`, just the cwd when the context is unknown, empty for an unusable payload. */
function summaryOf(input) {
  if (!input || typeof input !== 'object') return '';
  const cwd = typeof input.cwd === 'string' ? input.cwd : typeof input.workspace?.current_dir === 'string' ? input.workspace.current_dir : '';
  if (!cwd) return '';
  const p = input.context_window?.used_percentage;
  return isNum(p) ? `${cwd} · ctx ${Math.round(p)}%` : cwd;
}

function previousCommand(dir, account) {
  try {
    const state = JSON.parse(readFileSync(join(dir, 'statusline.json'), 'utf8'));
    const prev = state?.[account]?.previous;
    return prev && typeof prev.command === 'string' && prev.command ? prev.command : undefined;
  } catch {
    return undefined;
  }
}

function main() {
  const account = process.argv[2] ?? '';
  const dir = dirname(fileURLToPath(import.meta.url));
  const raw = readStdin();
  try {
    store(dir, account, JSON.parse(raw));
  } catch {
    /* the status line must never fail: unparsable input or an unwritable folder only skips the usage file */
  }
  const command = previousCommand(dir, account);
  if (!command) {
    try {
      const line = summaryOf(JSON.parse(raw));
      if (line) process.stdout.write(line);
    } catch {
      /* nothing to print */
    }
    return;
  }
  const res = spawnSync(command, { input: raw, shell: true, encoding: 'utf8', timeout: CHAIN_TIMEOUT_MS, windowsHide: true });
  if (res.stdout) process.stdout.write(res.stdout);
}

main();
